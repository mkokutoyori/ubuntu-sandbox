/**
 * SHA-256 (FIPS 180-4) — a faithful, dependency-free implementation.
 *
 * Pure 32-bit integer arithmetic via `Math.imul` and `>>> 0` so it behaves
 * identically in every JS engine. Operates on `Uint8Array` end to end.
 */
 
import { utf8ToBytes, bytesToHex } from '../encoding';
import type { HashAlgorithm } from './HashAlgorithm';
 
const BLOCK_SIZE = 64; // 512 bits
const DIGEST_SIZE = 32; // 256 bits

const rotr = (x: number, n: number): number => (x >>> n) | (x << (32 - n));
 
/** Compute the raw 32-byte SHA-256 digest of `input` (input is not mutated). */
export function sha256(input: Uint8Array): Uint8Array {
  // Initial hash values: fractional parts of the square roots of the first
  // eight primes (FIPS 180-4 §5.3.3).
  let h0 = 0x6a09e667, h1 = 0xbb67ae85, h2 = 0x3c6ef372, h3 = 0xa54ff53a;
  let h4 = 0x510e527f, h5 = 0x9b05688c, h6 = 0x1f83d9ab, h7 = 0x5be0cd19;
 
  const padded = padMessage(input);
  const w = new Uint32Array(64);
 
  for (let offset = 0; offset < padded.length; offset += BLOCK_SIZE) {
    for (let i = 0; i < 16; i++) {
      const j = offset + i * 4;
      w[i] = (padded[j] << 24) | (padded[j + 1] << 16) | (padded[j + 2] << 8) | padded[j + 3];
    }
    for (let i = 16; i < 64; i++) {
      const s0 = rotr(w[i - 15], 7) ^ rotr(w[i - 15], 18) ^ (w[i - 15] >>> 3);
      const s1 = rotr(w[i - 2], 17) ^ rotr(w[i - 2], 19) ^ (w[i - 2] >>> 10);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) >>> 0;
    }
 
    let a = h0, b = h1, c = h2, d = h3, e = h4, f = h5, g = h6, h = h7;
    for (let i = 0; i < 64; i++) {
      const S1 = rotr(e, 6) ^ rotr(e, 11) ^ rotr(e, 25);
      const ch = (e & f) ^ (~e & g);
      const temp1 = (h + S1 + ch + K[i] + w[i]) >>> 0;
      const S0 = rotr(a, 2) ^ rotr(a, 13) ^ rotr(a, 22);
      const maj = (a & b) ^ (a & c) ^ (b & c);
      const temp2 = (S0 + maj) >>> 0;
      h = g; g = f; f = e; e = (d + temp1) >>> 0;
      d = c; c = b; b = a; a = (temp1 + temp2) >>> 0;
    }
 
    h0 = (h0 + a) >>> 0; h1 = (h1 + b) >>> 0; h2 = (h2 + c) >>> 0; h3 = (h3 + d) >>> 0;
    h4 = (h4 + e) >>> 0; h5 = (h5 + f) >>> 0; h6 = (h6 + g) >>> 0; h7 = (h7 + h) >>> 0;
  }
 
  return wordsToBytes([h0, h1, h2, h3, h4, h5, h6, h7]);
}
 
/** Hex digest of a UTF-8 string — the common convenience path. */
export function sha256Hex(text: string): string {
  return bytesToHex(sha256(utf8ToBytes(text)));
}
 
/** Algorithm descriptor for HMAC / PBKDF2 (Dependency Inversion). */
export const SHA256: HashAlgorithm = {
  blockSize: BLOCK_SIZE,
  digestSize: DIGEST_SIZE,
  digest: sha256,
};
 
/** Apply the SHA-2 length padding (the 0x80 byte, zero fill, 64-bit length). */
function padMessage(input: Uint8Array): Uint8Array {
  const bitLen = input.length * 8;
  // One 0x80 byte, then pad with zeros until length ≡ 56 (mod 64), then 8-byte length.
  const paddedLen = ((input.length + 8) >> 6) * 64 + 64;
  const padded = new Uint8Array(paddedLen);
  padded.set(input);
  padded[input.length] = 0x80;
  // 64-bit big-endian bit length. JS bitwise ops are 32-bit, so split the word.
  const hi = Math.floor(bitLen / 0x100000000);
  const lo = bitLen >>> 0;
  const dv = new DataView(padded.buffer);
  dv.setUint32(paddedLen - 8, hi);
  dv.setUint32(paddedLen - 4, lo);
  return padded;
}
 
function wordsToBytes(words: number[]): Uint8Array {
  const out = new Uint8Array(words.length * 4);
  for (let i = 0; i < words.length; i++) {
    out[i * 4] = (words[i] >>> 24) & 0xff;
    out[i * 4 + 1] = (words[i] >>> 16) & 0xff;
    out[i * 4 + 2] = (words[i] >>> 8) & 0xff;
    out[i * 4 + 3] = words[i] & 0xff;
  }
  return out;
}
